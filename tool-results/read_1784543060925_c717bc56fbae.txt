     2→

    29→
    30→import {
    31→  Card,
    32→  CardContent,
    33→  CardDescription,

    35→  CardTitle,
    36→} from "@/components/ui/card";
    37→import { Button } from "@/components/ui/button";
    38→import { Input } from "@/components/ui/input";
    39→import { Badge } from "@/components/ui/badge";
    40→import { Skeleton } from "@/components/ui/skeleton";
    41→import { Checkbox } from "@/components/ui/checkbox";
    42→import {
    43→  Select,
    44→  SelectContent,
    45→  SelectItem,
    46→  SelectTrigger,
    47→  SelectValue,
    48→} from "@/components/ui/select";
    49→import {
    50→  Table,
    51→  TableBody,
    52→  TableCell,
    53→  TableHead,
    54→  TableHeader,
    55→  TableRow,
    56→} from "@/components/ui/table";
    57→
    58→import { StarRating } from "./star-rating";
    59→import { EmptyState } from "./empty-state";
    60→import { cleanReviewerName } from "@/lib/gbp/format";
    61→import type {
    62→  BranchesResponse,
    63→  BranchWithStats,
    64→  Review,

    67→
    68→interface ReviewsSectionProps {
    69→  /** Bump to force a refetch (e.g. after a manual scrape). */
    70→  refreshKey?: number;
    71→}
    72→
    73→const PAGE_SIZE_OPTIONS = [10, 25, 50] as const;
    74→const RATING_OPTIONS = [1, 2, 3, 4, 5] as const;
    75→
    76→/** A single review row, enriched with display names. */
    77→interface ReviewRow extends Review {
    78→  reviewer_display: string;
    79→  text_short: string;
    80→  competitor_name: string;
    81→  branch_name: string;
    82→}
    83→
    84→export function ReviewsSection({ refreshKey }: ReviewsSectionProps) {
    85→  // ── Filter state ─────────────────────────────────────────────────────────
    86→  const [branchId, setBranchId] = React.useState<string>("all");
    87→  const [competitorId, setCompetitorId] = React.useState<string>("all");
    88→  const [selectedRatings, setSelectedRatings] = React.useState<Set<number>>(
    89→    new Set(),
    90→  );
    91→  const [search, setSearch] = React.useState("");
    92→  const [debouncedSearch, setDebouncedSearch] = React.useState("");
    93→
    94→  // ── Pagination state (server-driven) ────────────────────────────────────
    95→  const [page, setPage] = React.useState(1);
    96→  const [pageSize, setPageSize] = React.useState<(typeof PAGE_SIZE_OPTIONS)[number]>(25);
    97→
    98→  // ── Sort state (client-side, applied to the current page) ───────────────
    99→  const [sorting, setSorting] = React.useState<SortingState>([]);
   100→  const [columnVisibility] = React.useState<VisibilityState>({});
   101→
   102→  // ── Data ────────────────────────────────────────────────────────────────
   103→  const [branches, setBranches] = React.useState<BranchesResponse | null>(null);
   104→  const [rows, setRows] = React.useState<ReviewRow[]>([]);
   105→  const [total, setTotal] = React.useState(0);
   106→  const [loading, setLoading] = React.useState(true);
   107→  const [error, setError] = React.useState<string | null>(null);
   108→
   109→  // Reset to page 1 when any filter changes.
   110→  React.useEffect(() => {
   111→    setPage(1);
   112→  }, [branchId, competitorId, selectedRatings, debouncedSearch, pageSize]);
   113→
   114→  // Debounce search input.
   115→  React.useEffect(() => {
   116→    const t = setTimeout(() => setDebouncedSearch(search), 300);
   117→    return () => clearTimeout(t);
   118→  }, [search]);
   119→
   120→  // Fetch branches tree once (for branch/competitor select options + name lookup).
   121→  React.useEffect(() => {
   122→    let cancelled = false;
   123→    fetch("/api/branches")
   124→      .then(async (r) => {
   125→        if (!r.ok) throw new Error(`HTTP ${r.status}`);
   126→        const json = (await r.json()) as BranchesResponse;
   127→        if (!cancelled) setBranches(json);
   128→      })
   129→      .catch(() => {
   130→        /* best-effort — names will just be IDs */
   131→      });
   132→    return () => {
   133→      cancelled = true;
   134→    };
   135→  }, [refreshKey]);
   136→
   137→  // Build a lookup map for competitor_id → {competitor_name, branch_name}.
   138→  const nameLookup = React.useMemo(() => {
   139→    const m = new Map<string, { competitor_name: string; branch_name: string }>();
   140→    for (const b of branches?.branches ?? []) {
   141→      for (const c of b.competitors) {
   142→        m.set(c.competitor_id, {
   143→          competitor_name: c.name,
   144→          branch_name: b.branch_name,
   145→        });
   146→      }
   147→    }
   148→    return m;
   149→  }, [branches]);
   150→
   151→  // Build the competitor options for the currently-selected branch.
   152→  const competitorOptions = React.useMemo(() => {
   153→    if (!branches) return [];
   154→    if (branchId === "all") {
   155→      // Flatten all competitors across all branches.
   156→      return branches.branches.flatMap((b: BranchWithStats) =>
   157→        b.competitors.map((c) => ({
   158→          competitor_id: c.competitor_id,
   159→          name: c.name,
   160→          branch_name: b.branch_name,
   161→        })),
   162→      );
   163→    }
   164→    const branch = branches.branches.find((b) => b.branch_id === branchId);
   165→    return (branch?.competitors ?? []).map((c) => ({
   166→      competitor_id: c.competitor_id,
   167→      name: c.name,
   168→      branch_name: branch?.branch_name ?? "",
   169→    }));
   170→  }, [branches, branchId]);
   171→
   172→  // Fetch reviews whenever filters / page / refreshKey change.
   173→  React.useEffect(() => {
   174→    let cancelled = false;
   175→    const controller = new AbortController();
   176→    setLoading(true);
   177→    setError(null);
   178→
   179→    const params = new URLSearchParams();
   180→    if (branchId !== "all") params.set("branch_id", branchId);
   181→    if (competitorId !== "all") params.set("competitor_id", competitorId);
   182→    if (selectedRatings.size > 0) {
   183→      params.set("rating", Array.from(selectedRatings).sort().join(","));
   184→    }
   185→    if (debouncedSearch.trim()) params.set("q", debouncedSearch.trim());
   186→    params.set("page", String(page));
   187→    params.set("pageSize", String(pageSize));
   188→
   189→    fetch(`/api/reviews?${params.toString()}`, { signal: controller.signal })
   190→      .then(async (r) => {
   191→        if (!r.ok) throw new Error(`HTTP ${r.status}`);
   192→        const json = (await r.json()) as ReviewsResponse;
   193→        if (cancelled) return;
   194→        const enriched: ReviewRow[] = json.data.map((rv) => {
   195→          const names = nameLookup.get(rv.competitor_id);
   196→          return {
   197→            ...rv,
   198→            reviewer_display: cleanReviewerName(rv.reviewer_name),
   199→            text_short: rv.text ?? "",
   200→            competitor_name: names?.competitor_name ?? rv.competitor_id,
   201→            branch_name: names?.branch_name ?? rv.branch_id,
   202→          };
   203→        });
   204→        setRows(enriched);
   205→        setTotal(json.total);
   206→      })
   207→      .catch((e) => {
   208→        if (cancelled) return;
   209→        if (e.name === "AbortError") return;
   210→        setError(e instanceof Error ? e.message : String(e));
   211→        setRows([]);
   212→        setTotal(0);
   213→      })
   214→      .finally(() => {
   215→        if (!cancelled) setLoading(false);
   216→      });
   217→
   218→    return () => {
   219→      cancelled = true;
   220→      controller.abort();
   221→    };
   222→  }, [
   223→    branchId,
   224→    competitorId,
   225→    selectedRatings,
   226→    debouncedSearch,
   227→    page,
   228→    pageSize,
   229→    refreshKey,
   230→    nameLookup,
   231→  ]);
   232→
   233→  // When branch changes, reset competitor selection (it may no longer be valid).
   234→  React.useEffect(() => {
   235→    setCompetitorId("all");
   236→  }, [branchId]);
   237→
   238→  // Toggle a rating in the multi-select.
   239→  const toggleRating = (r: number) => {
   240→    setSelectedRatings((prev) => {
   241→      const next = new Set(prev);
   242→      if (next.has(r)) next.delete(r);
   243→      else next.add(r);
   244→      return next;
   245→    });
   246→  };
   247→
   248→  // ── Table definition ────────────────────────────────────────────────────
   249→  const columns = React.useMemo<ColumnDef<ReviewRow>[]>(
   250→    () => [
   251→      {
   252→        accessorKey: "reviewer_display",
   253→        header: "Reviewer",
   254→        cell: ({ row }) => (
   255→          <div className="flex flex-col gap-0.5">
   256→            <span className="font-medium text-foreground">
   257→              {row.original.reviewer_display}
   258→            </span>
   259→            <span className="font-mono text-[10px] text-muted-foreground">
   260→              {row.original.review_id}
   261→            </span>
   262→          </div>
   263→        ),
   264→        sortingFn: "alphanumeric",
   265→      },
   266→      {
   267→        accessorKey: "rating",
   268→        header: "Rating",
   269→        cell: ({ row }) => <StarRating rating={row.original.rating} size="sm" />,
   270→        sortingFn: "basic",
   271→      },
   272→      {
   273→        accessorKey: "text_short",
   274→        header: "Review",
   275→        enableSorting: false,
   276→        cell: ({ row }) => <ReviewTextCell text={row.original.text} />,
   277→      },
   278→      {
   279→        accessorKey: "relative_date",
   280→        header: "When",
   281→        cell: ({ row }) => (
   282→          <span className="text-xs text-muted-foreground">
   283→            {row.original.relative_date ?? "—"}
   284→          </span>
   285→        ),
   286→        sortingFn: "alphanumeric",
   287→      },
   288→      {
   289→        accessorKey: "competitor_name",
   290→        header: "Competitor",
   291→        cell: ({ row }) => (
   292→          <div className="flex flex-col gap-0.5">
   293→            <span className="text-xs font-medium text-foreground">
   294→              {row.original.competitor_name}
   295→            </span>
   296→            <span className="text-[10px] text-muted-foreground">
   297→              {row.original.branch_name}
   298→            </span>
   299→          </div>
   300→        ),
   301→        sortingFn: "alphanumeric",
   302→      },
   303→    ],
   304→    [],
   305→  );
   306→
   307→  const table = useReactTable({
   308→    data: rows,
   309→    columns,
   310→    state: { sorting, columnVisibility },
   311→    onSortingChange: setSorting,
   312→    getCoreRowModel: getCoreRowModel(),
   313→    getSortedRowModel: getSortedRowModel(),
   314→    getFilteredRowModel: getFilteredRowModel(),
   315→    getPaginationRowModel: getPaginationRowModel(),
   316→    // Pagination is server-driven; we let the table think all rows fit on the
   317→    // current page so its internal pagination doesn't fight ours.
   318→    pageCount: 1,
   319→    manualPagination: true,
   320→  });
   321→
   322→  const totalPages = Math.max(1, Math.ceil(total / pageSize));
   323→  const canPrev = page > 1;
   324→  const canNext = page < totalPages;
   325→
   326→  const hasActiveFilters =
   327→    branchId !== "all" ||
   328→    competitorId !== "all" ||
   329→    selectedRatings.size > 0 ||
   330→    debouncedSearch.trim().length > 0;
   331→
   332→  const clearFilters = () => {
   333→    setBranchId("all");
   334→    setCompetitorId("all");
   335→    setSelectedRatings(new Set());
   336→    setSearch("");
   337→  };
   338→
   339→  return (
   340→    <motion.div
   341→      initial={{ opacity: 0, y: 8 }}
   342→      animate={{ opacity: 1, y: 0 }}
   343→      transition={{ duration: 0.3, ease: "easeOut" }}
   344→      className="space-y-4"
   345→    >
   346→      <Card className="gbp-card-hover bg-gradient-to-br from-primary/5 to-transparent">
   347→        <CardHeader>
   348→          <CardTitle className="flex items-center gap-2 text-base">
   349→            <MessageSquare className="size-4 text-primary" aria-hidden="true" />
   350→            All Reviews
   351→          </CardTitle>
   352→          <CardDescription>
   353→            Searchable, filterable, paginated view across every competitor snapshot.
   354→            {" "}
   355→            <span className="font-semibold text-foreground">
   356→              {total.toLocaleString()}
   357→            </span>{" "}
   358→            review{total === 1 ? "" : "s"} match the current filters.
   359→          </CardDescription>
   360→        </CardHeader>
   361→        <CardContent>
   362→          {/* Filter bar */}
   363→          <div className="grid grid-cols-1 gap-3 md:grid-cols-2 lg:grid-cols-4">
   364→            <div className="space-y-1.5">
   365→              <label className="text-[11px] font-medium uppercase tracking-wide text-muted-foreground">
   366→                Branch
   367→              </label>
   368→              <Select value={branchId} onValueChange={setBranchId}>
   369→                <SelectTrigger className="w-full" size="sm">
   370→                  <SelectValue placeholder="All branches" />
   371→                </SelectTrigger>
   372→                <SelectContent>
   373→                  <SelectItem value="all">All branches</SelectItem>
   374→                  {branches?.branches.map((b) => (
   375→                    <SelectItem key={b.branch_id} value={b.branch_id}>
   376→                      {b.branch_name}
   377→                    </SelectItem>
   378→                  ))}
   379→                </SelectContent>
   380→              </Select>
   381→            </div>
   382→
   383→            <div className="space-y-1.5">
   384→              <label className="text-[11px] font-medium uppercase tracking-wide text-muted-foreground">
   385→                Competitor
   386→              </label>
   387→              <Select value={competitorId} onValueChange={setCompetitorId}>
   388→                <SelectTrigger className="w-full" size="sm">
   389→                  <SelectValue placeholder="All competitors" />
   390→                </SelectTrigger>
   391→                <SelectContent>
   392→                  <SelectItem value="all">All competitors</SelectItem>
   393→                  {competitorOptions.map((c) => (
   394→                    <SelectItem key={c.competitor_id} value={c.competitor_id}>
   395→                      {c.name}
   396→                    </SelectItem>
   397→                  ))}
   398→                </SelectContent>
   399→              </Select>
   400→            </div>
   401→
   402→            <div className="space-y-1.5">
   403→              <label className="text-[11px] font-medium uppercase tracking-wide text-muted-foreground">
   404→                Rating
   405→              </label>
   406→              <div className="flex h-8 items-center gap-1.5 rounded-md border border-input bg-transparent px-2">
   407→                {RATING_OPTIONS.map((r) => {
   408→                  const active = selectedRatings.has(r);
   409→                  return (
   410→                    <button
   411→                      key={r}
   412→                      type="button"
   413→                      onClick={() => toggleRating(r)}
   414→                      aria-pressed={active}
   415→                      aria-label={`Filter by ${r} star${r === 1 ? "" : "s"}`}
   416→                      className={`inline-flex items-center gap-0.5 rounded px-1.5 py-0.5 text-xs font-medium transition-colors ${
   417→                        active
   418→                          ? "bg-primary text-primary-foreground"
   419→                          : "text-muted-foreground hover:bg-accent hover:text-accent-foreground"
   420→                      }`}
   421→                    >
   422→                      {r}
   423→                      <Star
   424→                        className={`size-3 ${
   425→                          active
   426→                            ? "fill-primary-foreground text-primary-foreground"
   427→                            : "fill-amber-400 text-amber-500"
   428→                        }`}
   429→                        aria-hidden="true"
   430→                      />
   431→                    </button>
   432→                  );
   433→                })}
   434→              </div>
   435→            </div>
   436→
   437→            <div className="space-y-1.5">
   438→              <label className="text-[11px] font-medium uppercase tracking-wide text-muted-foreground">
   439→                Search
   440→              </label>
   441→              <div className="relative">
   442→                <Search
   443→                  className="pointer-events-none absolute left-2.5 top-1/2 size-4 -translate-y-1/2 text-muted-foreground"
   444→                  aria-hidden="true"
   445→                />
   446→                <Input
   447→                  value={search}
   448→                  onChange={(e) => setSearch(e.target.value)}
   449→                  placeholder="Reviewer or text…"
   450→                  className="h-8 pl-8 text-sm"
   451→                  aria-label="Search reviews"
   452→                />
   453→                {search && (
   454→                  <button
   455→                    type="button"
   456→                    onClick={() => setSearch("")}
   457→                    aria-label="Clear search"
   458→                    className="absolute right-1.5 top-1/2 -translate-y-1/2 rounded p-0.5 text-muted-foreground hover:text-foreground"
   459→                  >
   460→                    <X className="size-3.5" aria-hidden="true" />
   461→                  </button>
   462→                )}
   463→              </div>
   464→            </div>
   465→          </div>
   466→
   467→          {hasActiveFilters && (
   468→            <div className="mt-3 flex items-center justify-between">
   469→              <div className="flex flex-wrap items-center gap-2 text-xs text-muted-foreground">
   470→                <span>Active filters:</span>
   471→                {branchId !== "all" && (
   472→                  <Badge variant="secondary" className="gap-1 text-[11px]">
   473→                    Branch:{" "}
   474→                    {branches?.branches.find((b) => b.branch_id === branchId)
   475→                      ?.branch_name ?? branchId}
   476→                  </Badge>
   477→                )}
   478→                {competitorId !== "all" && (
   479→                  <Badge variant="secondary" className="gap-1 text-[11px]">
   480→                    Competitor:{" "}
   481→                    {competitorOptions.find((c) => c.competitor_id === competitorId)
   482→                      ?.name ?? competitorId}
   483→                  </Badge>
   484→                )}
   485→                {selectedRatings.size > 0 && (
   486→                  <Badge variant="secondary" className="gap-1 text-[11px]">
   487→                    Ratings: {Array.from(selectedRatings).sort().join(", ")}★
   488→                  </Badge>
   489→                )}
   490→                {debouncedSearch.trim() && (
   491→                  <Badge variant="secondary" className="gap-1 text-[11px]">
   492→                    “{debouncedSearch.trim()}”
   493→                  </Badge>
   494→                )}
   495→              </div>
   496→              <Button
   497→                variant="ghost"
   498→                size="sm"
   499→                onClick={clearFilters}
   500→                className="h-7 text-xs text-muted-foreground hover:text-foreground"
   501→              >
   502→                <X className="size-3" aria-hidden="true" />
   503→                Clear
   504→              </Button>
   505→            </div>
   506→          )}
   507→        </CardContent>
   508→      </Card>
   509→
   510→      {/* Table */}
   511→      <Card className="gbp-card-hover">
   512→        <CardContent className="p-0">
   513→          <div className="max-h-[70vh] overflow-y-auto gbp-scrollbar-lg">
   514→            <Table>
   515→              <TableHeader className="sticky top-0 z-10 bg-card shadow-sm">
   516→                {table.getHeaderGroups().map((hg) => (
   517→                  <TableRow key={hg.id} className="hover:bg-transparent">
   518→                    {hg.headers.map((header) => {
   519→                      const sortable = header.column.getCanSort();
   520→                      const sorted = header.column.getIsSorted();
   521→                      return (
   522→                        <TableHead
   523→                          key={header.id}
   524→                          className="bg-card px-3 text-xs font-semibold uppercase tracking-wide"
   525→                        >
   526→                          {header.isPlaceholder ? null : sortable ? (
   527→                            <button
   528→                              type="button"
   529→                              onClick={header.column.getToggleSortingHandler()}
   530→                              className="inline-flex items-center gap-1 hover:text-primary"
   531→                            >
   532→                              {flexRender(
   533→                                header.column.columnDef.header,
   534→                                header.getContext(),
   535→                              )}
   536→                              {sorted === "asc" ? (
   537→                                <ArrowUp className="size-3" aria-hidden="true" />
   538→                              ) : sorted === "desc" ? (
   539→                                <ArrowDown className="size-3" aria-hidden="true" />
   540→                              ) : (
   541→                                <ArrowUpDown
   542→                                  className="size-3 opacity-40"
   543→                                  aria-hidden="true"
   544→                                />
   545→                              )}
   546→                            </button>
   547→                          ) : (
   548→                            flexRender(
   549→                              header.column.columnDef.header,
   550→                              header.getContext(),
   551→                            )
   552→                          )}
   553→                        </TableHead>
   554→                      );
   555→                    })}
   556→                  </TableRow>
   557→                ))}
   558→              </TableHeader>
   559→              <TableBody>
   560→                {loading ? (
   561→                  Array.from({ length: 6 }).map((_, i) => (
   562→                    <TableRow key={`sk-${i}`} className="hover:bg-transparent">
   563→                      {columns.map((_, j) => (
   564→                        <TableCell key={j} className="px-3 py-3">
   565→                          <Skeleton className="h-5 w-full max-w-[180px]" />
   566→                        </TableCell>
   567→                      ))}
   568→                    </TableRow>
   569→                  ))
   570→                ) : rows.length === 0 ? (
   571→                  <TableRow className="hover:bg-transparent">
   572→                    <TableCell colSpan={columns.length} className="py-10">
   573→                      <EmptyState
   574→                        icon={hasActiveFilters ? Inbox : MessageSquare}
   575→                        title={
   576→                          hasActiveFilters
   577→                            ? "No reviews match these filters"
   578→                            : "No reviews yet"
   579→                        }
   580→                        description={
   581→                          hasActiveFilters
   582→                            ? "Try widening your filters or clearing them."
   583→                            : "The scraper hasn't produced any snapshots yet. Click “Run Now” in the header to trigger a fixtures-mode scrape."
   584→                        }
   585→                      />
   586→                    </TableCell>
   587→                  </TableRow>
   588→                ) : error ? (
   589→                  <TableRow className="hover:bg-transparent">
   590→                    <TableCell colSpan={columns.length} className="py-10">
   591→                      <EmptyState
   592→                        icon={AlertTriangle}
   593→                        title="Couldn't load reviews"
   594→                        description={error}
   595→                      />
   596→                    </TableCell>
   597→                  </TableRow>
   598→                ) : (
   599→                  table.getRowModel().rows.map((row, idx) => (
   600→                    <TableRow
   601→                      key={row.original.review_id}
   602→                      className={idx % 2 === 1 ? "bg-muted/30" : ""}
   603→                    >
   604→                      {row.getVisibleCells().map((cell) => (
   605→                        <TableCell key={cell.id} className="px-3 align-top">
   606→                          {flexRender(
   607→                            cell.column.columnDef.cell,
   608→                            cell.getContext(),
   609→                          )}
   610→                        </TableCell>
   611→                      ))}
   612→                    </TableRow>
   613→                  ))
   614→                )}
   615→              </TableBody>
   616→            </Table>
   617→          </div>
   618→
   619→          {/* Pagination footer */}
   620→          <div className="flex flex-col items-center justify-between gap-3 border-t border-border/60 px-4 py-3 sm:flex-row">
   621→            <div className="flex items-center gap-3 text-xs text-muted-foreground">
   622→              <span>
   623→                Showing{" "}
   624→                <span className="font-semibold tabular-nums text-foreground">
   625→                  {rows.length === 0 ? 0 : (page - 1) * pageSize + 1}
   626→                </span>
   627→                –
   628→                <span className="font-semibold tabular-nums text-foreground">
   629→                  {Math.min(page * pageSize, total)}
   630→                </span>{" "}
   631→                of{" "}
   632→                <span className="font-semibold tabular-nums text-foreground">
   633→                  {total.toLocaleString()}
   634→                </span>
   635→              </span>
   636→              <span className="text-border">·</span>
   637→              <span className="inline-flex items-center gap-1.5">
   638→                Per page:
   639→                <Select
   640→                  value={String(pageSize)}
   641→                  onValueChange={(v) =>
   642→                    setPageSize(Number(v) as (typeof PAGE_SIZE_OPTIONS)[number])
   643→                  }
   644→                >
   645→                  <SelectTrigger size="sm" className="h-7 w-[68px] text-xs">
   646→                    <SelectValue />
   647→                  </SelectTrigger>
   648→                  <SelectContent>
   649→                    {PAGE_SIZE_OPTIONS.map((n) => (
   650→                      <SelectItem key={n} value={String(n)}>
   651→                        {n}
   652→                      </SelectItem>
   653→                    ))}
   654→                  </SelectContent>
   655→                </Select>
   656→              </span>
   657→            </div>
   658→            <div className="flex items-center gap-2">
   659→              <Button
   660→                variant="outline"
   661→                size="sm"
   662→                onClick={() => setPage((p) => Math.max(1, p - 1))}
   663→                disabled={!canPrev || loading}
   664→                aria-label="Previous page"
   665→              >
   666→                <ChevronLeft className="size-4" aria-hidden="true" />
   667→                Prev
   668→              </Button>
   669→              <span className="text-xs text-muted-foreground tabular-nums">
   670→                Page{" "}
   671→                <span className="font-semibold text-foreground">{page}</span> /{" "}
   672→                {totalPages}
   673→              </span>
   674→              <Button
   675→                variant="outline"
   676→                size="sm"
   677→                onClick={() => setPage((p) => Math.min(totalPages, p + 1))}
   678→                disabled={!canNext || loading}
   679→                aria-label="Next page"
   680→              >
   681→                Next
   682→                <ChevronRight className="size-4" aria-hidden="true" />
   683→              </Button>
   684→            </div>
   685→          </div>
   686→        </CardContent>
   687→      </Card>
   688→    </motion.div>
   689→  );
   690→}

   692→/** Review text cell with click-to-expand for long text. */
   693→function ReviewTextCell({ text }: { text: string | null }) {
   694→  const [expanded, setExpanded] = React.useState(false);
   695→  if (!text) {
   696→    return <span className="text-xs italic text-muted-foreground">(no text)</span>;
   697→  }
   698→  const MAX = 120;
   699→  if (text.length <= MAX) {
   700→    return <span className="text-sm leading-relaxed text-foreground/90">{text}</span>;
   701→  }
   702→  return (
   703→    <div className="text-sm leading-relaxed text-foreground/90">
   704→      {expanded ? text : `${text.slice(0, MAX).trimEnd()}…`}
   705→      <button
   706→        type="button"
   707→        onClick={() => setExpanded((v) => !v)}
   708→        className="ml-1 text-xs font-medium text-primary hover:underline focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-ring rounded"
   709→        aria-expanded={expanded}
   710→      >
   711→        {expanded ? "show less" : "expand"}
   712→      </button>
   713→    </div>
   714→  );
   715→}
