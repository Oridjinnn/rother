    83→    83→
    84→    84→export function ReviewsSection({ refreshKey }: ReviewsSectionProps) {
    85→    85→  // ── Filter state ─────────────────────────────────────────────────────────
    86→    86→  const [branchId, setBranchId] = React.useState<string>("all");
    87→    87→  const [competitorId, setCompetitorId] = React.useState<string>("all");
    88→    88→  const [selectedRatings, setSelectedRatings] = React.useState<Set<number>>(
    89→    89→    new Set(),
    90→    90→  );
    91→    91→  const [search, setSearch] = React.useState("");
    92→    92→  const [debouncedSearch, setDebouncedSearch] = React.useState("");
    93→    93→
    94→    94→  // ── Pagination state (server-driven) ────────────────────────────────────
    95→    95→  const [page, setPage] = React.useState(1);
    96→    96→  const [pageSize, setPageSize] = React.useState<(typeof PAGE_SIZE_OPTIONS)[number]>(25);
    97→    97→
    98→    98→  // ── Sort state (client-side, applied to the current page) ───────────────
    99→    99→  const [sorting, setSorting] = React.useState<SortingState>([]);
   100→   100→  const [columnVisibility] = React.useState<VisibilityState>({});
   101→   101→
   102→   102→  // ── Data ────────────────────────────────────────────────────────────────
   103→   103→  const [branches, setBranches] = React.useState<BranchesResponse | null>(null);
   104→   104→  const [rows, setRows] = React.useState<ReviewRow[]>([]);
   105→   105→  const [total, setTotal] = React.useState(0);
   106→   106→  const [loading, setLoading] = React.useState(true);
   107→   107→  const [error, setError] = React.useState<string | null>(null);
   108→   108→
   109→   109→  // Reset to page 1 when any filter changes.
   110→   110→  React.useEffect(() => {
   111→   111→    setPage(1);
   112→   112→  }, [branchId, competitorId, selectedRatings, debouncedSearch, pageSize]);
   113→   113→
   114→   114→  // Debounce search input.
   115→   115→  React.useEffect(() => {
   116→   116→    const t = setTimeout(() => setDebouncedSearch(search), 300);
   117→   117→    return () => clearTimeout(t);
   118→   118→  }, [search]);
   119→   119→
   120→   120→  // Fetch branches tree once (for branch/competitor select options + name lookup).
   121→   121→  React.useEffect(() => {
   122→   122→    let cancelled = false;
   123→   123→    fetch("/api/branches")
   124→   124→      .then(async (r) => {
   125→   125→        if (!r.ok) throw new Error(`HTTP ${r.status}`);
   126→   126→        const json = (await r.json()) as BranchesResponse;
   127→   127→        if (!cancelled) setBranches(json);
   128→   128→      })
   129→   129→      .catch(() => {
   130→   130→        /* best-effort — names will just be IDs */
   131→   131→      });
   132→   132→    return () => {
   133→   133→      cancelled = true;
   134→   134→    };
   135→   135→  }, [refreshKey]);
   136→   136→
   137→   137→  // Build a lookup map for competitor_id → {competitor_name, branch_name}.
   138→   138→  const nameLookup = React.useMemo(() => {
   139→   139→    const m = new Map<string, { competitor_name: string; branch_name: string }>();

   141→   141→      for (const c of b.competitors) {
   142→   142→        m.set(c.competitor_id, {
   143→   143→          competitor_name: c.name,
   144→   144→          branch_name: b.branch_name,
   145→   145→        });
   146→   146→      }
   147→   147→    }
   148→   148→    return m;
   149→   149→  }, [branches]);
   150→   150→
   151→   151→  // Build the competitor options for the currently-selected branch.
   152→   152→  const competitorOptions = React.useMemo(() => {
   153→   153→    if (!branches) return [];
   154→   154→    if (branchId === "all") {
   155→   155→      // Flatten all competitors across all branches.
   156→   156→      return branches.branches.flatMap((b: BranchWithStats) =>
   157→   157→        b.competitors.map((c) => ({
   158→   158→          competitor_id: c.competitor_id,
   159→   159→          name: c.name,
   160→   160→          branch_name: b.branch_name,
   161→   161→        })),
   162→   162→      );
   163→   163→    }
   164→   164→    const branch = branches.branches.find((b) => b.branch_id === branchId);
   165→   165→    return (branch?.competitors ?? []).map((c) => ({
   166→   166→      competitor_id: c.competitor_id,
   167→   167→      name: c.name,
   168→   168→      branch_name: branch?.branch_name ?? "",
   169→   169→    }));
   170→   170→  }, [branches, branchId]);
   171→   171→
   172→   172→  // Fetch reviews whenever filters / page / refreshKey change.
   173→   173→  React.useEffect(() => {
   174→   174→    let cancelled = false;
   175→   175→    const controller = new AbortController();
   176→   176→    setLoading(true);
   177→   177→    setError(null);
   178→   178→
   179→   179→    const params = new URLSearchParams();
   180→   180→    if (branchId !== "all") params.set("branch_id", branchId);
   181→   181→    if (competitorId !== "all") params.set("competitor_id", competitorId);
   182→   182→    if (selectedRatings.size > 0) {
   183→   183→      params.set("rating", Array.from(selectedRatings).sort().join(","));
   184→   184→    }
   185→   185→    if (debouncedSearch.trim()) params.set("q", debouncedSearch.trim());
   186→   186→    params.set("page", String(page));
   187→   187→    params.set("pageSize", String(pageSize));
   188→   188→
   189→   189→    fetch(`/api/reviews?${params.toString()}`, { signal: controller.signal })
   190→   190→      .then(async (r) => {
   191→   191→        if (!r.ok) throw new Error(`HTTP ${r.status}`);
   192→   192→        const json = (await r.json()) as ReviewsResponse;
   193→   193→        if (cancelled) return;
   194→   194→        const enriched: ReviewRow[] = json.data.map((rv) => {
   195→   195→          const names = nameLookup.get(rv.competitor_id);
   196→   196→          return {
   197→   197→            ...rv,
   198→   198→            reviewer_display: cleanReviewerName(rv.reviewer_name),
   199→   199→            text_short: rv.text ?? "",
   200→   200→            competitor_name: names?.competitor_name ?? rv.competitor_id,
   201→   201→            branch_name: names?.branch_name ?? rv.branch_id,
   202→   202→          };
   203→   203→        });
   204→   204→        setRows(enriched);
   205→   205→        setTotal(json.total);
   206→   206→      })
   207→   207→      .catch((e) => {
   208→   208→        if (cancelled) return;
   209→   209→        if (e.name === "AbortError") return;
   210→   210→        setError(e instanceof Error ? e.message : String(e));
   211→   211→        setRows([]);
   212→   212→        setTotal(0);
   213→   213→      })
   214→   214→      .finally(() => {
   215→   215→        if (!cancelled) setLoading(false);
   216→   216→      });
   217→   217→
   218→   218→    return () => {
   219→   219→      cancelled = true;
   220→   220→      controller.abort();
   221→   221→    };
   222→   222→  }, [
   223→   223→    branchId,
   224→   224→    competitorId,
   225→   225→    selectedRatings,
   226→   226→    debouncedSearch,
   227→   227→    page,
   228→   228→    pageSize,
   229→   229→    refreshKey,
   230→   230→    nameLookup,
   231→   231→  ]);
   232→   232→
   233→   233→  // When branch changes, reset competitor selection (it may no longer be valid).
   234→   234→  React.useEffect(() => {
   235→   235→    setCompetitorId("all");
   236→   236→  }, [branchId]);
   237→   237→
   238→   238→  // Toggle a rating in the multi-select.
   239→   239→  const toggleRating = (r: number) => {
   240→   240→    setSelectedRatings((prev) => {
   241→   241→      const next = new Set(prev);
   242→   242→      if (next.has(r)) next.delete(r);
   243→   243→      else next.add(r);
   244→   244→      return next;
   245→   245→    });
   246→   246→  };
   247→   247→
   248→   248→  // ── Table definition ────────────────────────────────────────────────────
   249→   249→  const columns = React.useMemo<ColumnDef<ReviewRow>[]>(
   250→   250→    () => [
   251→   251→      {
   252→   252→        accessorKey: "reviewer_display",
   253→   253→        header: "Reviewer",
   254→   254→        cell: ({ row }) => (
   255→   255→          <div className="flex flex-col gap-0.5">
   256→   256→            <span className="font-medium text-foreground">
   257→   257→              {row.original.reviewer_display}
   258→   258→            </span>
   259→   259→            <span className="font-mono text-[10px] text-muted-foreground">
   260→   260→              {row.original.review_id}
   261→   261→            </span>
   262→   262→          </div>
   263→   263→        ),
   264→   264→        sortingFn: "alphanumeric",
   265→   265→      },
   266→   266→      {
   267→   267→        accessorKey: "rating",
   268→   268→        header: "Rating",
   269→   269→        cell: ({ row }) => <StarRating rating={row.original.rating} size="sm" />,
   270→   270→        sortingFn: "basic",
   271→   271→      },
   272→   272→      {
   273→   273→        accessorKey: "text_short",
   274→   274→        header: "Review",
   275→   275→        enableSorting: false,
   276→   276→        cell: ({ row }) => <ReviewTextCell text={row.original.text} />,
   277→   277→      },
   278→   278→      {
   279→   279→        accessorKey: "relative_date",
   280→   280→        header: "When",
   281→   281→        cell: ({ row }) => (
   282→   282→          <span className="text-xs text-muted-foreground">
   283→   283→            {row.original.relative_date ?? "—"}
   284→   284→          </span>
   285→   285→        ),
   286→   286→        sortingFn: "alphanumeric",
   287→   287→      },
   288→   288→      {
   289→   289→        accessorKey: "competitor_name",
   290→   290→        header: "Competitor",
   291→   291→        cell: ({ row }) => (
   292→   292→          <div className="flex flex-col gap-0.5">
   293→   293→            <span className="text-xs font-medium text-foreground">
   294→   294→              {row.original.competitor_name}
   295→   295→            </span>
   296→   296→            <span className="text-[10px] text-muted-foreground">
   297→   297→              {row.original.branch_name}
   298→   298→            </span>
   299→   299→          </div>
   300→   300→        ),
   301→   301→        sortingFn: "alphanumeric",
   302→   302→      },
   303→   303→    ],
   304→   304→    [],
   305→   305→  );
   306→   306→
   307→   307→  const table = useReactTable({
   308→   308→    data: rows,
   309→   309→    columns,
   310→   310→    state: { sorting, columnVisibility },
   311→   311→    onSortingChange: setSorting,
   312→   312→    getCoreRowModel: getCoreRowModel(),
   313→   313→    getSortedRowModel: getSortedRowModel(),
   314→   314→    getFilteredRowModel: getFilteredRowModel(),
   315→   315→    getPaginationRowModel: getPaginationRowModel(),
   316→   316→    // Pagination is server-driven; we let the table think all rows fit on the
   317→   317→    // current page so its internal pagination doesn't fight ours.
   318→   318→    pageCount: 1,
   319→   319→    manualPagination: true,
   320→   320→  });
   321→   321→
   322→   322→  const totalPages = Math.max(1, Math.ceil(total / pageSize));
   323→   323→  const canPrev = page > 1;
   324→   324→  const canNext = page < totalPages;
   325→   325→
   326→   326→  const hasActiveFilters =
   327→   327→    branchId !== "all" ||
   328→   328→    competitorId !== "all" ||
   329→   329→    selectedRatings.size > 0 ||
   330→   330→    debouncedSearch.trim().length > 0;
   331→   331→
   332→   332→  const clearFilters = () => {
   333→   333→    setBranchId("all");
   334→   334→    setCompetitorId("all");
   335→   335→    setSelectedRatings(new Set());
   336→   336→    setSearch("");
   337→   337→  };
   338→   338→
   339→   339→  return (
   340→   340→    <motion.div
   341→   341→      initial={{ opacity: 0, y: 8 }}
   342→   342→      animate={{ opacity: 1, y: 0 }}
   343→   343→      transition={{ duration: 0.3, ease: "easeOut" }}
   344→   344→      className="space-y-4"
   345→   345→    >
   346→   346→      <Card className="gbp-card-hover bg-gradient-to-br from-primary/5 to-transparent">
   347→   347→        <CardHeader>
   348→   348→          <CardTitle className="flex items-center gap-2 text-base">
   349→   349→            <MessageSquare className="size-4 text-primary" aria-hidden="true" />
   350→   350→            All Reviews
   351→   351→          </CardTitle>
   352→   352→          <CardDescription>
   353→   353→            Searchable, filterable, paginated view across every competitor snapshot.
   354→   354→            {" "}
   355→   355→            <span className="font-semibold text-foreground">
   356→   356→              {total.toLocaleString()}
   357→   357→            </span>{" "}
   358→   358→            review{total === 1 ? "" : "s"} match the current filters.
   359→   359→          </CardDescription>
   360→   360→        </CardHeader>
   361→   361→        <CardContent>
   362→   362→          {/* Filter bar */}
   363→   363→          <div className="grid grid-cols-1 gap-3 md:grid-cols-2 lg:grid-cols-4">
   364→   364→            <div className="space-y-1.5">
   365→   365→              <label className="text-[11px] font-medium uppercase tracking-wide text-muted-foreground">
   366→   366→                Branch
   367→   367→              </label>
   368→   368→              <Select value={branchId} onValueChange={setBranchId}>
   369→   369→                <SelectTrigger className="w-full" size="sm">
   370→   370→                  <SelectValue placeholder="All branches" />
   371→   371→                </SelectTrigger>
   372→   372→                <SelectContent>
   373→   373→                  <SelectItem value="all">All branches</SelectItem>
   374→   374→                  {branches?.branches.map((b) => (
   375→   375→                    <SelectItem key={b.branch_id} value={b.branch_id}>
   376→   376→                      {b.branch_name}
   377→   377→                    </SelectItem>
   378→   378→                  ))}
   379→   379→                </SelectContent>
   380→   380→              </Select>
   381→   381→            </div>
   382→   382→
   383→   383→            <div className="space-y-1.5">
   384→   384→              <label className="text-[11px] font-medium uppercase tracking-wide text-muted-foreground">
   385→   385→                Competitor
   386→   386→              </label>
   387→   387→              <Select value={competitorId} onValueChange={setCompetitorId}>
   388→   388→                <SelectTrigger className="w-full" size="sm">
   389→   389→                  <SelectValue placeholder="All competitors" />
   390→   390→                </SelectTrigger>
   391→   391→                <SelectContent>
   392→   392→                  <SelectItem value="all">All competitors</SelectItem>
   393→   393→                  {competitorOptions.map((c) => (
   394→   394→                    <SelectItem key={c.competitor_id} value={c.competitor_id}>
   395→   395→                      {c.name}
   396→   396→                    </SelectItem>
   397→   397→                  ))}
   398→   398→                </SelectContent>
   399→   399→              </Select>
   400→   400→            </div>
   401→   401→
   402→   402→            <div className="space-y-1.5">
   403→   403→              <label className="text-[11px] font-medium uppercase tracking-wide text-muted-foreground">
   404→   404→                Rating
   405→   405→              </label>
   406→   406→              <div className="flex h-8 items-center gap-1.5 rounded-md border border-input bg-transparent px-2">
   407→   407→                {RATING_OPTIONS.map((r) => {
   408→   408→                  const active = selectedRatings.has(r);
   409→   409→                  return (
   410→   410→                    <button
   411→   411→                      key={r}
   412→   412→                      type="button"
   413→   413→                      onClick={() => toggleRating(r)}
   414→   414→                      aria-pressed={active}
   415→   415→                      aria-label={`Filter by ${r} star${r === 1 ? "" : "s"}`}
   416→   416→                      className={`inline-flex items-center gap-0.5 rounded px-1.5 py-0.5 text-xs font-medium transition-colors ${
   417→   417→                        active
   418→   418→                          ? "bg-primary text-primary-foreground"
   419→   419→                          : "text-muted-foreground hover:bg-accent hover:text-accent-foreground"
   420→   420→                      }`}
   421→   421→                    >
   422→   422→                      {r}
   423→   423→                      <Star
   424→   424→                        className={`size-3 ${
   425→   425→                          active
   426→   426→                            ? "fill-primary-foreground text-primary-foreground"
   427→   427→                            : "fill-amber-400 text-amber-500"
   428→   428→                        }`}
   429→   429→                        aria-hidden="true"
   430→   430→                      />
   431→   431→                    </button>
   432→   432→                  );
   433→   433→                })}
   434→   434→              </div>
   435→   435→            </div>
   436→   436→
   437→   437→            <div className="space-y-1.5">
   438→   438→              <label className="text-[11px] font-medium uppercase tracking-wide text-muted-foreground">
   439→   439→                Search
   440→   440→              </label>
   441→   441→              <div className="relative">
   442→   442→                <Search
   443→   443→                  className="pointer-events-none absolute left-2.5 top-1/2 size-4 -translate-y-1/2 text-muted-foreground"
   444→   444→                  aria-hidden="true"
   445→   445→                />
   446→   446→                <Input
   447→   447→                  value={search}
   448→   448→                  onChange={(e) => setSearch(e.target.value)}
   449→   449→                  placeholder="Reviewer or text…"
   450→   450→                  className="h-8 pl-8 text-sm"
   451→   451→                  aria-label="Search reviews"
   452→   452→                />
   453→   453→                {search && (
   454→   454→                  <button
   455→   455→                    type="button"
   456→   456→                    onClick={() => setSearch("")}
   457→   457→                    aria-label="Clear search"
   458→   458→                    className="absolute right-1.5 top-1/2 -translate-y-1/2 rounded p-0.5 text-muted-foreground hover:text-foreground"
   459→   459→                  >
   460→   460→                    <X className="size-3.5" aria-hidden="true" />
   461→   461→                  </button>
   462→   462→                )}
   463→   463→              </div>
   464→   464→            </div>
   465→   465→          </div>
   466→   466→
   467→   467→          {hasActiveFilters && (
   468→   468→            <div className="mt-3 flex items-center justify-between">
   469→   469→              <div className="flex flex-wrap items-center gap-2 text-xs text-muted-foreground">
   470→   470→                <span>Active filters:</span>
   471→   471→                {branchId !== "all" && (
   472→   472→                  <Badge variant="secondary" className="gap-1 text-[11px]">
   473→   473→                    Branch:{" "}
   474→   474→                    {branches?.branches.find((b) => b.branch_id === branchId)
   475→   475→                      ?.branch_name ?? branchId}
   476→   476→                  </Badge>
   477→   477→                )}
   478→   478→                {competitorId !== "all" && (
   479→   479→                  <Badge variant="secondary" className="gap-1 text-[11px]">
   480→   480→                    Competitor:{" "}
   481→   481→                    {competitorOptions.find((c) => c.competitor_id === competitorId)
   482→   482→                      ?.name ?? competitorId}
   483→   483→                  </Badge>
   484→   484→                )}
   485→   485→                {selectedRatings.size > 0 && (

   502→   502→                <X className="size-3" aria-hidden="true" />

   510→   510→      {/* Table */}
   511→   511→      <Card className="gbp-card-hover">
   512→   512→        <CardContent className="p-0">
   513→   513→          <div className="max-h-[70vh] overflow-y-auto gbp-scrollbar-lg">

   557→   557→                ))}
   558→   558→              </TableHeader>
   559→   559→              <TableBody>
   560→   560→                {loading ? (
   561→   561→                  Array.from({ length: 6 }).map((_, i) => (

   576→   576→                          hasActiveFilters
   577→   577→                            ? "No reviews match these filters"
   578→   578→                            : "No reviews yet"
   579→   579→                        }

   581→   581→                          hasActiveFilters

   656→   656→              </span>

   658→   658→            <div className="flex items-center gap-2">
   659→   659→              <Button
   660→   660→                variant="outline"

   662→   662→                onClick={() => setPage((p) => Math.max(1, p - 1))}
   663→   663→                disabled={!canPrev || loading}
   664→   664→                aria-label="Previous page"
   665→   665→              >
   666→   666→                <ChevronLeft className="size-4" aria-hidden="true" />
   667→   667→                Prev
   668→   668→              </Button>
   669→   669→              <span className="text-xs text-muted-foreground tabular-nums">
   670→   670→                Page{" "}
   671→   671→                <span className="font-semibold text-foreground">{page}</span> /{" "}
   672→   672→                {totalPages}
   673→   673→              </span>
   674→   674→              <Button
   675→   675→                variant="outline"
   676→   676→                size="sm"
   677→   677→                onClick={() => setPage((p) => Math.min(totalPages, p + 1))}
   678→   678→                disabled={!canNext || loading}
   679→   679→                aria-label="Next page"
   680→   680→              >
   681→   681→                Next
   682→   682→                <ChevronRight className="size-4" aria-hidden="true" />
   683→   683→              </Button>
   684→   684→            </div>
   685→   685→          </div>
   686→   686→        </CardContent>
   687→   687→      </Card>
   688→   688→    </motion.div>
   689→   689→  );
   690→   690→}

   692→   692→/** Review text cell with click-to-expand for long text. */
   693→   693→function ReviewTextCell({ text }: { text: string | null }) {
   694→   694→  const [expanded, setExpanded] = React.useState(false);
   695→   695→  if (!text) {
   696→   696→    return <span className="text-xs italic text-muted-foreground">(no text)</span>;
   697→   697→  }
   698→   698→  const MAX = 120;
   699→   699→  if (text.length <= MAX) {
   700→   700→    return <span className="text-sm leading-relaxed text-foreground/90">{text}</span>;
   701→   701→  }
   702→   702→  return (
   703→   703→    <div className="text-sm leading-relaxed text-foreground/90">
   704→   704→      {expanded ? text : `${text.slice(0, MAX).trimEnd()}…`}
   705→   705→      <button
   706→   706→        type="button"
   707→   707→        onClick={() => setExpanded((v) => !v)}
   708→   708→        className="ml-1 text-xs font-medium text-primary hover:underline focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-ring rounded"
   709→   709→        aria-expanded={expanded}
   710→   710→      >
   711→   711→        {expanded ? "show less" : "expand"}
   712→   712→      </button>
   713→   713→    </div>
   714→   714→  );
   715→   715→}
